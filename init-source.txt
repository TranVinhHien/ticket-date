docker run -d \
  --name ticket_postgres \
  --restart unless-stopped \
  --network ticket_network \
  -p 5432:5432 \
  -e POSTGRES_USER=postgres \
  -e POSTGRES_PASSWORD=postgres \
  -e POSTGRES_DB=ticket \
  -e PGDATA=/var/lib/postgresql/data/pgdata \
  -v "$(pwd)/be/data/postgres:/var/lib/postgresql/data" \
  postgres:16-alpine

docker run -d \
  --name ticket_redis \
  --restart unless-stopped \
  --network ticket_network \
  -p 6379:6379 \
  -v "$(pwd)/be/data/redis:/data" \
  redis:7-alpine redis-server --appendonly yes
